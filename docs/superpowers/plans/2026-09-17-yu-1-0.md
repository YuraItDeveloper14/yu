# Yu 1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Yu 1.0.0: one version in the CLI and the Studio, `yu` for Windows, Linux and macOS with checksums, a repository with the files contributors look for, security headers on the site, and an accessibility check in CI.

**Architecture:** The workspace version in `Cargo.toml` is the only source of the version: the CLI prints it, the core hands it to the Studio through `names()`, and the release workflow checks the tag against it. `.github/workflows/release.yml` builds and runs the three binaries on their own runners and publishes them with `gh release create`. The site's headers live in `studio/public/vercel.json`, and `vite preview` reads that file, so every browser test runs under the policy the site sends; axe runs as more browser tests.

**Tech Stack:** Rust stable (a workspace with no dependencies), GitHub Actions (`actions/checkout@v7`, `actions/setup-node@v7`, `actions/upload-artifact@v7`, `actions/download-artifact@v8`, `dtolnay/rust-toolchain@stable`), Vite 8.3.0, TypeScript 7.0.2, Playwright 1.63.0, `@axe-core/playwright` 4.13.0, Vercel.

### Task 2: The release workflow

**Files:**
- Create: `.github/workflows/release.yml`

**Interfaces:**
- Consumes: the workspace version from `Cargo.toml`; `examples/hello.yu` and `examples/hello.out`; the `## [<version>]` section of `CHANGELOG.md` (Task 1).
- Produces: on a tag `v<version>`, a GitHub release with `yu-windows-x64.zip`, `yu-linux-x64.tar.gz`, `yu-macos.tar.gz` and `SHA256SUMS.txt` (Task 3's README links to them).

- [ ] **Step 1: Write the workflow**

Create `.github/workflows/release.yml`:

```yaml
name: release

# A pushed tag v1.2.3 builds yu for Windows, Linux and macOS, runs each binary, and publishes the
# release. A pull request that changes this file builds and runs the same binaries without publishing.
on:
  push:
    tags: ['v*']
  pull_request:
    paths: ['.github/workflows/release.yml']
  workflow_dispatch:

permissions:
  contents: read

jobs:
  version:
    runs-on: ubuntu-latest
    outputs:
      version: ${{ steps.read.outputs.version }}
    steps:
      - uses: actions/checkout@v7
      - id: read
        name: The workspace version
        run: echo "version=$(grep -m1 '^version' Cargo.toml | cut -d'"' -f2)" >> "$GITHUB_OUTPUT"
      - name: A tag names the same version
        if: github.ref_type == 'tag'
        run: test "$GITHUB_REF_NAME" = "v${{ steps.read.outputs.version }}"

  windows:
    needs: version
    runs-on: windows-latest
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v7
      - uses: dtolnay/rust-toolchain@stable
      - run: cargo build --release --locked -p yu-cli
      - name: The binary runs
        run: |
          test "$(target/release/yu.exe --version)" = "yu ${{ needs.version.outputs.version }}"
          diff <(target/release/yu.exe examples/hello.yu) examples/hello.out
      - name: Pack
        shell: pwsh
        run: |
          New-Item -ItemType Directory yu | Out-Null
          Copy-Item target/release/yu.exe, README.md, LICENSE yu
          Copy-Item examples yu -Recurse
          Compress-Archive -Path yu -DestinationPath yu-windows-x64.zip
      - uses: actions/upload-artifact@v7
        with:
          name: yu-windows-x64
          path: yu-windows-x64.zip
          if-no-files-found: error

  linux:
    needs: version
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: dtolnay/rust-toolchain@stable
        with:
          targets: x86_64-unknown-linux-musl
      - run: cargo build --release --locked -p yu-cli --target x86_64-unknown-linux-musl
      - name: The binary runs and needs no shared libraries
        run: |
          yu=target/x86_64-unknown-linux-musl/release/yu
          test "$($yu --version)" = "yu ${{ needs.version.outputs.version }}"
          diff <($yu examples/hello.yu) examples/hello.out
          if readelf -d "$yu" | grep -q NEEDED; then echo "yu needs shared libraries" >&2; exit 1; fi
      - name: Pack
        run: |
          mkdir yu
          cp target/x86_64-unknown-linux-musl/release/yu README.md LICENSE yu/
          cp -r examples yu/
          tar -czf yu-linux-x64.tar.gz yu
      - uses: actions/upload-artifact@v7
        with:
          name: yu-linux-x64
          path: yu-linux-x64.tar.gz
          if-no-files-found: error

  macos:
    needs: version
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v7
      - uses: dtolnay/rust-toolchain@stable
        with:
          targets: aarch64-apple-darwin, x86_64-apple-darwin
      - name: Build for both processors and join them
        run: |
          cargo build --release --locked -p yu-cli --target aarch64-apple-darwin
          cargo build --release --locked -p yu-cli --target x86_64-apple-darwin
          lipo -create -output yu-universal target/aarch64-apple-darwin/release/yu target/x86_64-apple-darwin/release/yu
      - name: The binary runs and holds both processors
        run: |
          test "$(./yu-universal --version)" = "yu ${{ needs.version.outputs.version }}"
          diff <(./yu-universal examples/hello.yu) examples/hello.out
          lipo -archs yu-universal | grep -q x86_64
          lipo -archs yu-universal | grep -q arm64
      - name: Pack
        run: |
          mkdir yu
          cp yu-universal yu/yu
          cp README.md LICENSE yu/
          cp -r examples yu/
          tar -czf yu-macos.tar.gz yu
      - uses: actions/upload-artifact@v7
        with:
          name: yu-macos
          path: yu-macos.tar.gz
          if-no-files-found: error

  publish:
    if: github.ref_type == 'tag'
    needs: [windows, linux, macos]
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v7
      - uses: actions/download-artifact@v8
        with:
          path: dist
          merge-multiple: true
      - name: Checksums
        working-directory: dist
        run: |
          sha256sum yu-* > SHA256SUMS.txt
          cat SHA256SUMS.txt
      - name: Notes from the changelog
        run: |
          awk -v head="## [${GITHUB_REF_NAME#v}]" 'index($0, head) == 1 { on = 1; next } /^## \[/ { on = 0 } on && !/^\[[^]]+\]: /' CHANGELOG.md > notes.md
          test -s notes.md
      - name: Publish
        env:
          GH_TOKEN: ${{ github.token }}
        run: gh release create "$GITHUB_REF_NAME" dist/* --title "Yu ${GITHUB_REF_NAME#v}" --notes-file notes.md
```

import { expect, test } from '@playwright/test';

export function watchEachPage(): void {
  let problems: string[] = [];

  test.beforeEach(async ({ page }) => {
    problems = [];
    page.on('pageerror', (error) => problems.push(error.message));
    page.on('console', (message) => {
      if (message.text().includes('Content Security Policy')) problems.push(message.text());
    });
    await page.addInitScript(() => {
      document.addEventListener('securitypolicyviolation', (event) => {
        console.error(`Content Security Policy blocked ${event.blockedURI} (${event.violatedDirective})`);
      });
    });
  });

  test.afterEach(() => {
    expect(problems).toEqual([]);
  });
}

/** The `--accent-soft: rgba(r, g, b, a)` of the block that starts with `selector {`. */
function softOf(selector: string): number[] {
  const start = css.indexOf(`${selector} {`);
  const block = css.slice(start, css.indexOf('}', start));
  const match = /--accent-soft:\s*rgba\((\d+),\s*(\d+),\s*(\d+),\s*([\d.]+)\)/.exec(block);
  assert.ok(match, `no --accent-soft in ${selector}`);
  return match.slice(1).map(Number);
}

/** A highlighted row: `soft` laid over the background `under`. */
function over(soft: number[], under: string): string {
  const [r, g, b, alpha] = soft;
  return `#${[r, g, b]
    .map((channel, i) => {
      const below = parseInt(under.slice(1 + i * 2, 3 + i * 2), 16);
      return Math.round(alpha * channel + (1 - alpha) * below).toString(16).padStart(2, '0');
    })
    .join('')}`;
}

const soft: Record<string, number[]> = {
  dark: softOf(':root'),
  light: softOf(":root[data-theme='light']"),
};
